Quick Start

Python (Sync)

from nexora_rs import NexoraClient

client = NexoraClient("http://localhost:8080")

# Create a node with properties
client.set_property("alice", "name", "Alice")
client.set_property("alice", "age", 32)

# Read properties
name = client.get_property("alice", "name")
print(name)  # "Alice"

# Add edges
client.add_edge("alice", "bob", "KNOWS")

# Query with Cypher
results = client.query("MATCH (n) RETURN n.name, n.age LIMIT 10")
for row in results:
    print(row)

# Register a standing query
sq = client.create_standing_query(
    name="high-temp",
    pattern={"type": "PropertyFilter", "key": "temp",
             "condition": {"type": "GreaterThan", "value": 10}}
)
print(f"Registered SQ: {sq['id']}")

client.close()

Python (Async)

import asyncio
from nexora_rs import AsyncNexoraClient

async def main():
    client = AsyncNexoraClient("http://localhost:8080")

    # Batch operations
    await client.set_property("alice", "name", "Alice")
    await client.set_property("bob", "name", "Bob")
    await client.add_edge("alice", "bob", "KNOWS")

    # Concurrent queries
    results = await client.query("MATCH (n) RETURN n LIMIT 10")
    print(results)

    await client.close()

asyncio.run(main())

TypeScript

import { NexoraClient } from 'nexora';

const client = new NexoraClient('http://localhost:8080');

// Create nodes
await client.setProperty('alice', 'name', 'Alice');
await client.setProperty('alice', 'age', 32);

// Query
const rows = await client.query('MATCH (n) RETURN n LIMIT 10');
console.log(rows);

// Standing queries
const sq = await client.createStandingQuery({
  name: 'high-temp',
  pattern: { type: 'PropertyFilter', key: 'temp',
             condition: { type: 'GreaterThan', value: 10 } }
});
console.log(`Registered: ${sq.id}`);

Rust

use nexora_client::NexoraClient;

let client = NexoraClient::new("http://localhost:8080")?;

// Create a node
client.set_property("alice", "name", "Alice").await?;

// Query
let results = client.query("MATCH (n) RETURN n LIMIT 10").await?;
for row in &results.rows {
    println!("{:?}", row);
}

Python Async Client API

Same API as the sync client, but all methods are async and return coroutines.

Streaming (WebSocket)

from nexora_rs import AsyncNexoraClient
from nexora_rs.streaming import StreamingClient

async def main():
    client = AsyncNexoraClient("http://localhost:8080")
    streaming = StreamingClient(client)

    # Subscribe to standing query matches
    async for event in streaming.subscribe():
        print(f"Event: {event}")

        if event.type == "sq_match":
            print(f"  Match: {event.data}")

# Run with reconnect support
import asyncio
asyncio.run(main())
